import mammoth from 'mammoth';

export async function extractTextFromFile(file: File): Promise<{ text: string; warnings: string[] }> {
  const name = file.name.toLowerCase();
  const warnings: string[] = [];

  // DOCX
  if (name.endsWith('.docx')) {
    const arrayBuffer = await file.arrayBuffer();
    const res = await mammoth.extractRawText({ arrayBuffer });
    const text = (res.value || '').trim();
    if (!text) warnings.push('DOCX extracted empty text');
    return { text, warnings };
  }

  // Plain text-like
  if (name.endsWith('.txt') || name.endsWith('.md') || name.endsWith('.markdown') || name.endsWith('.json')) {
    const text = (await file.text()).trim();
    if (!text) warnings.push('Uploaded file is empty');
    return { text, warnings };
  }

  // PDF not supported (UI-only)
  if (name.endsWith('.pdf')) {
    warnings.push('PDF parsing not supported in UI demo. Please export Workspaces output as DOCX.');
    return { text: '', warnings };
  }

  // Fallback
  warnings.push('Unknown file type. Treating as text.');
  const text = (await file.text()).trim();
  return { text, warnings };
}

export type StylusTemplateSections = {
  rawText: string;
  epicTitle: string;
  descriptionText: string;        // section 1 content
  inScopeText: string;            // section 2 content
  acceptanceCriteriaText: string; // section 4 content
  warnings: string[];
};

function normalize(raw: string) {
  return raw.replace(/\r\n/g, '\n').replace(/[ \t]+/g, ' ').replace(/\n{3,}/g, '\n\n').trim();
}

function sliceSection(text: string, startRe: RegExp, endRes: RegExp[]) {
  const m = text.match(startRe);
  if (!m || m.index == null) return null;
  const startIdx = m.index + m[0].length;
  let endIdx = text.length;
  for (const re of endRes) {
    const mm = text.slice(startIdx).match(re);
    if (mm && mm.index != null) endIdx = Math.min(endIdx, startIdx + mm.index);
  }
  return text.slice(startIdx, endIdx).trim();
}

export function parseStylusTemplateText(rawText: string): StylusTemplateSections {
  const warnings: string[] = [];
  const text = normalize(rawText);

  let epicTitle = '';
  const epicMatch = text.match(/^EPIC:\s*(.+)$/im);
  if (epicMatch?.[1]) epicTitle = epicMatch[1].trim();
  if (!epicTitle) {
    epicTitle = (text.split('\n').find(l => l.trim()) || 'Untitled Epic').trim();
    warnings.push('EPIC title line not found; using first non-empty line.');
  }

  const descriptionText = sliceSection(
    text,
    /^1\.\s*Description.*$/im,
    [/^2\.\s*In Scope.*$/im, /^3\.\s*/im, /^4\.\s*Acceptance Criteria.*$/im]
  ) || '';

  const inScopeText = sliceSection(
    text,
    /^2\.\s*In Scope.*$/im,
    [/^3\.\s*/im, /^4\.\s*Acceptance Criteria.*$/im]
  ) || '';

  const acceptanceCriteriaText = sliceSection(
    text,
    /^4\.\s*Acceptance Criteria.*$/im,
    [/^5\.\s*/im]
  ) || '';

  if (!descriptionText) warnings.push('Section "1. Description" not found.');
  if (!inScopeText) warnings.push('Section "2. In Scope" not found.');
  if (!acceptanceCriteriaText) warnings.push('Section "4. Acceptance Criteria" not found.');

  return { rawText: text, epicTitle, descriptionText, inScopeText, acceptanceCriteriaText, warnings };
}

import { CAPABILITIES, type CapabilityId } from './capabilities';
import type { StylusTemplateSections } from './stylusTemplateParser';

const RULES: Array<{ id: CapabilityId; keywords: string[] }> = [
  { id: 'posting' as CapabilityId, keywords: ['posting', 'posting restriction', 'no debit', 'no credit', 'account status'] },
  { id: 'clearing' as CapabilityId, keywords: ['clearing', 'settlement', 'clearing system'] },
  { id: 'sanctions' as CapabilityId, keywords: ['sanction', 'screening', 'ofac'] },
  { id: 'liquidity' as CapabilityId, keywords: ['liquidity', 'funding', 'position'] },
  { id: 'data' as CapabilityId, keywords: ['mongo', 'schema', 'data store', 'database', 'collection'] },
  { id: 'state-manager' as CapabilityId, keywords: ['state manager', 'fsm', 'orchestrator', 'workflow'] },
  { id: 'payment-initiation' as CapabilityId, keywords: ['payment initiation', 'initiation', 'pacs', 'pain'] },
  { id: 'validation' as CapabilityId, keywords: ['validate', 'validation', 'account check', 'dup check'] },
];

export function detectCapabilityFromStylusDoc(sections: StylusTemplateSections): {
  capabilityId: CapabilityId | null;
  confidence: number;
  reason: string;
} {
  const hay = `${sections.epicTitle}\n${sections.descriptionText}\n${sections.inScopeText}\n${sections.acceptanceCriteriaText}`.toLowerCase();

  // direct label match
  for (const c of CAPABILITIES) {
    if (hay.includes(c.label.toLowerCase())) {
      return { capabilityId: c.id as CapabilityId, confidence: 0.92, reason: `Matched capability label "${c.label}".` };
    }
  }

  let best: { id: CapabilityId; score: number; hit: string } | null = null;
  for (const r of RULES) {
    let score = 0;
    let hit = '';
    for (const k of r.keywords) {
      if (hay.includes(k)) {
        score += (k.split(' ').length >= 2 ? 2 : 1);
        if (!hit) hit = k;
      }
    }
    if (!best || score > best.score) best = { id: r.id, score, hit };
  }

  if (!best || best.score < 2) {
    return { capabilityId: null, confidence: 0.0, reason: 'No reliable keyword match.' };
  }

  const confidence = Math.min(0.55 + best.score * 0.08, 0.9);
  return { capabilityId: best.id, confidence, reason: `Keyword match: "${best.hit}".` };
}

function extractBullets(text: string): string[] {
  return text
    .split('\n')
    .map(l => l.trim())
    .filter(Boolean)
    .map(l => l.replace(/^[-•*–]\s+/, '').replace(/^\d+[\.\)]\s+/, ''))
    .filter(l => l.length > 2);
}

function fingerprint(country: string, capabilityId: string, epicTitle: string) {
  const norm = epicTitle.toLowerCase().replace(/[^a-z0-9 ]/g, '').replace(/\s+/g, ' ').trim();
  return `${(country || 'xx').toLowerCase()}|${capabilityId}|${norm}`;
}

function scopeHeuristic(text: string) {
  const t = text.toLowerCase();
  if (/(implement|develop|new endpoint|new api|code change|refactor)/.test(t)) return 'CODE_CHANGE';
  return 'CONFIG_ONLY';
}

export function buildJiraEpicFromStylusDoc(args: {

  const children = bullets.map((b, idx) => ({

  return {
    capabilityId: args.capabilityId,
    title: args.sections.epicTitle,

    components: meta.components,
    owner: { team: meta.ownerTeam, name: 'TBD' },
    children,
    fingerprint: fingerprint(country, String(args.capabilityId), args.sections.epicTitle),
  } as any;
}

import type { JiraEpicDraft } from './analysisTypes';

const KEY = 'jiraDraftRegistry.v1';

type Registry = Record<string, JiraEpicDraft>;

export function loadRegistry(): Registry {
  try { return JSON.parse(localStorage.getItem(KEY) || '{}') as Registry; } catch { return {}; }
}
export function saveRegistry(reg: Registry) {
  localStorage.setItem(KEY, JSON.stringify(reg));
}
export function upsertDraft(draft: JiraEpicDraft): { mode: 'create'|'update'; previous?: JiraEpicDraft } {
  const reg = loadRegistry();
  const fp = (draft as any).fingerprint;
  if (!fp) {
    const id = `no-fp-${Date.now()}`;
    (draft as any).fingerprint = id;
    reg[id] = draft;
    saveRegistry(reg);
    return { mode: 'create' };
  }
  const prev = reg[fp];
  reg[fp] = draft;
  saveRegistry(reg);
  return prev ? { mode: 'update', previous: prev } : { mode: 'create' };
}

export function diffLines(oldText: string, newText: string) {
  const oldLines = new Set(oldText.split('\n').map(l => l.trim()).filter(Boolean));
  const newLines = new Set(newText.split('\n').map(l => l.trim()).filter(Boolean));
  const added: string[] = [];
  const removed: string[] = [];
  for (const l of newLines) if (!oldLines.has(l)) added.push(l);
  for (const l of oldLines) if (!newLines.has(l)) removed.push(l);
  return { added, removed };
}
